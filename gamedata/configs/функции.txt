function npc_check_unique_item(corpse_id)
     local corpse=level.object_by_id(corpse_id)
     if corpse~=nil then
        for z,w in pairs(unique_items_list) do
            if corpse:object(w[1]) then
               result="true"
            else
               result="false"
            end
        end
     end
     return result
end
